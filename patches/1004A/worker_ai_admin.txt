async function handleAdminAi(request, env) {
  const origin = request.headers.get("Origin") || "*";
  const cors = {
    "Access-Control-Allow-Origin": env.ALLOWED_ORIGIN || origin,
    "Access-Control-Allow-Methods": "POST, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type",
    "Content-Type": "application/json; charset=utf-8"
  };
  if (request.method === "OPTIONS") return new Response(null, { status: 204, headers: cors });
  if (request.method !== "POST") return new Response(JSON.stringify({ error: "POST only" }), { status: 405, headers: cors });
  if (env.ALLOWED_ORIGIN && origin !== env.ALLOWED_ORIGIN) return new Response(JSON.stringify({ error: "origin" }), { status: 403, headers: cors });
  if (!env.ANTHROPIC_API_KEY) return new Response(JSON.stringify({ error: "no key" }), { status: 500, headers: cors });

  let b;
  try { b = await request.json(); } catch (e) { return new Response(JSON.stringify({ error: "bad json" }), { status: 400, headers: cors }); }
  const text = String(b.text || "").slice(0, 2000);
  const allowed = new Set((b.tools || []).map(t => t.name));
  const tools = ADMIN_AI_TOOLS.filter(t => allowed.size === 0 || allowed.has(t.name));
  const history = (b.history || []).slice(-8)
    .filter(m => m && m.text)
    .map(m => ({ role: m.role === "user" ? "user" : "assistant", content: String(m.text).slice(0, 2000) }));
  // Anthropic 要求第一則是 user、而且 user/assistant 交替；最後一則是這次的需求
  const msgs = [];
  for (const m of history) { if (!msgs.length && m.role !== "user") continue; if (msgs.length && msgs[msgs.length-1].role === m.role) msgs[msgs.length-1].content += "\n" + m.content; else msgs.push(m); }
  if (msgs.length && msgs[msgs.length-1].role === "user") msgs.pop();
  msgs.push({ role: "user", content: text });

  const system =
    "你是 KGM Airways 管理後台的 AI 助理。今天是 " + (b.today || "") + "。登入者職務：" + (b.role || "") + "，員工編號 " + (b.empId || "") + "。\n" +
    "把使用者的後台需求轉成工具呼叫；一次需求可以呼叫多個工具。日期一律用 YYYY-MM-DD（沒寫年份就用今天之後最近的那一天）。\n" +
    "權限由網站端依職務檢查，沒有權限的工具網站會回覆「無權限」，你不用自己判斷。\n" +
    "看不懂或缺資料時不要亂猜，直接用一句中文反問。回覆一律用繁體中文，簡短。";

  async function call(model) {
    const r = await fetch("https://api.anthropic.com/v1/messages", {
      method: "POST",
      headers: { "x-api-key": env.ANTHROPIC_API_KEY, "anthropic-version": "2023-06-01", "content-type": "application/json" },
      body: JSON.stringify({ model, max_tokens: 1024, system, tools, messages: msgs })
    });
    if (!r.ok) throw new Error("anthropic " + r.status + " " + (await r.text()).slice(0, 200));
    return r.json();
  }

  const primary = b.model || "claude-opus-5-5";
  const fallback = b.fallbackModel || "claude-sonnet-5";
  let j, used = primary;
  try { j = await call(primary); }
  catch (e1) {
    try { j = await call(fallback); used = fallback; }
    catch (e2) { return new Response(JSON.stringify({ error: String(e2.message || e2) }), { status: 502, headers: cors }); }
  }
  const reply = (j.content || []).filter(c => c.type === "text").map(c => c.text).join("\n").trim();
  const actions = (j.content || []).filter(c => c.type === "tool_use").map(c => ({ tool: c.name, args: c.input || {} }));
  return new Response(JSON.stringify({ reply, actions, model: used }), { headers: cors });
}
