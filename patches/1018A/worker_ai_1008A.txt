function corsOf(request, env) {
  const origin = request.headers.get("Origin") || "*";
  return {
    "Access-Control-Allow-Origin": env.ALLOWED_ORIGIN || origin,
    "Access-Control-Allow-Methods": "POST, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type",
    "Content-Type": "application/json; charset=utf-8"
  };
}
function historyToMessages(history, text) {
  const h = (history || []).slice(-8).filter(m => m && m.text)
    .map(m => ({ role: m.role === "user" ? "user" : "assistant", content: String(m.text).slice(0, 2000) }));
  // Anthropic 要求第一則是 user、user/assistant 交替；最後一則是這次的需求
  const msgs = [];
  for (const m of h) { if (!msgs.length && m.role !== "user") continue; if (msgs.length && msgs[msgs.length-1].role === m.role) msgs[msgs.length-1].content += "\n" + m.content; else msgs.push(m); }
  if (msgs.length && msgs[msgs.length-1].role === "user") msgs.pop();
  msgs.push({ role: "user", content: String(text || "").slice(0, 2000) });
  return msgs;
}
async function callClaude(env, model, system, tools, messages, maxTokens) {
  const body = { model, max_tokens: maxTokens || 1024, system, messages };
  if (tools && tools.length) body.tools = tools;
  const r = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: { "x-api-key": env.ANTHROPIC_API_KEY, "anthropic-version": "2023-06-01", "content-type": "application/json" },
    body: JSON.stringify(body)
  });
  if (!r.ok) throw new Error("anthropic " + r.status + " " + (await r.text()).slice(0, 200));
  return r.json();
}
async function callWithFallback(env, primary, fallback, system, tools, messages, maxTokens) {
  try { return { j: await callClaude(env, primary, system, tools, messages, maxTokens), used: primary }; }
  catch (e1) { return { j: await callClaude(env, fallback, system, tools, messages, maxTokens), used: fallback }; }
}

async function handleFrontAi(request, env) {
  const cors = corsOf(request, env), origin = request.headers.get("Origin") || "*";
  if (request.method === "OPTIONS") return new Response(null, { status: 204, headers: cors });
  if (request.method !== "POST") return new Response(JSON.stringify({ error: "POST only" }), { status: 405, headers: cors });
  if (env.ALLOWED_ORIGIN && origin !== env.ALLOWED_ORIGIN) return new Response(JSON.stringify({ error: "origin" }), { status: 403, headers: cors });
  if (!env.ANTHROPIC_API_KEY) return new Response(JSON.stringify({ error: "no key" }), { status: 500, headers: cors });
  let b;
  try { b = await request.json(); } catch (e) { return new Response(JSON.stringify({ error: "bad json" }), { status: 400, headers: cors }); }
  const en = b.lang === "en";
  const pol = (b.policy || []).slice(0, 3).map((p, i) => (i + 1) + ". " + String(p).slice(0, 1200)).join("\n");
  const mem = b.member ? ("旅客已登入：卡等 " + (b.member.tier || "—") + "，可用里程 " + (b.member.miles || 0) + "。") : "旅客未登入。";
  const sr = b.search && b.search.fr ? ("旅客目前的搜尋條件：" + b.search.fr + "→" + b.search.to + " " + (b.search.dep || "") + (b.search.ret ? " 回程 " + b.search.ret : "") + "。") : "";
  const system =
    "你是 KGM Airways 官網的客服小幫手。今天是 " + (b.today || "") + "。旅客正在看「" + (b.page || "home") + "」頁。" + mem + sr + "\n" +
    "回答依據只限下面的 KGM 規章條文與網站提供的資料；條文沒寫的（票價、剩餘座位、航班是否有位）不要猜，請旅客用 search_flights 打開搜尋結果看，或轉真人客服。\n" +
    "引用規定時說明是哪一份規定第幾條。旅客要查航班就呼叫 search_flights（日期沒寫年份就用今天之後最近的那一天；「下個月」「最便宜哪天」先挑下個月中旬的一天打開，提醒旅客可以在結果頁切換日期比價）。\n" +
    "旅客要找真人、或需要人工處理的事就呼叫 handoff_agent。不要承諾退款或賠償金額。\n" +
    (en ? "Reply in English, briefly." : "一律用繁體中文回答，口吻親切、簡短（三到五句）。") + "\n\n" +
    "【相關規章條文】\n" + (pol || "（沒有找到相關條文）");
  let out;
  try { out = await callWithFallback(env, b.model || "claude-sonnet-5-5", b.fallbackModel || "claude-haiku-4-5-20251001", system, FRONT_AI_TOOLS, historyToMessages(b.history, b.text), 600); }
  catch (e) { return new Response(JSON.stringify({ error: String(e.message || e) }), { status: 502, headers: cors }); }
  const j = out.j;
  const reply = (j.content || []).filter(c => c.type === "text").map(c => c.text).join("\n").trim();
  const tu = (j.content || []).filter(c => c.type === "tool_use")[0];
  let action = null;
  if (tu && tu.name === "search_flights") action = Object.assign({ type: "search" }, tu.input || {});
  if (tu && tu.name === "open_page") action = { type: "open", page: (tu.input || {}).page };
  if (tu && tu.name === "handoff_agent") action = { type: "handoff" };
  return new Response(JSON.stringify({ reply, action, model: out.used }), { headers: cors });
}
